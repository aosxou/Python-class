import random as r
import turtle as t

# ------------------------------------------------------------
# [중간 마지막 문제] - 산수 퀴즈 프로그램
# ------------------------------------------------------------
def quiz_game():
    while True:
        num1 = r.randint(10, 99)
        num2 = r.randint(10, 99)
        op = r.choice(['+', '-', '*', '/'])

        if op == '+':
            answer = num1 + num2
        elif op == '-':
            answer = num1 - num2
        elif op == '*':
            answer = num1 * num2
        elif op == '/':
            answer = num1 // num2  # 정수 나눗셈

        while True:
            print(f"\n출력: {num1} {op} {num2} = ")
            user = int(input("사용자 입력 : "))

            if user == answer:
                print("출력: 정답입니다. 산수퀴즈를 계속할까요?")
                again = input("사용자 입력 (Y,y 또는 N,n) : ")
                if again in ['n', 'N']:
                    print("프로그램을 종료합니다.")
                    return
                else:
                    break
            else:
                print("출력: 오답입니다.")


# ------------------------------------------------------------
# [실습문제 1] - 아이디 확인 및 3자리 비밀번호 생성
# ------------------------------------------------------------
def id_check():
    std_id = ['shingu', 'it', 'software']

    user = input("\n아이디 입력 : ")
    if user in std_id:
        print("환영합니다")
        pw = r.randint(100, 999)
        print(f"3자리 암호는 {pw}입니다.")
    else:
        print("없는 id입니다.")


# ------------------------------------------------------------
# [실습문제 2] - 학년별 수석 학생 구하기
# ------------------------------------------------------------
def top_students():
    num = int(input("\n입력 받을 학생 수 : "))

    max1, name1 = 0, ""
    max2, name2 = 0, ""
    max3, name3 = 0, ""
    total = 0

    for i in range(num):
        print(f"\n[{i + 1}번째 학생 입력]")
        name = input("영문 이니셜 : ")
        grade = int(input("학년(1~3) : "))
        score = int(input("총점(100점 만점) : "))
        total += score

        if grade == 1:
            if score > max1:
                max1 = score
                name1 = name
        elif grade == 2:
            if score > max2:
                max2 = score
                name2 = name
        elif grade == 3:
            if score > max3:
                max3 = score
                name3 = name

    print(f"\n1학년 수석 : {name1} 총점: {max1}점")
    print(f"2학년 수석 : {name2} 총점: {max2}점")
    print(f"3학년 수석 : {name3} 총점: {max3}점")

    avg = total / num
    print(f"\n총 {num}명 학생들의 평점 : {avg:.2f}점")


# ------------------------------------------------------------
# [실습문제 3] - 거북이 그래픽 (랜덤 도형 그리기)
# ------------------------------------------------------------
def draw_shapes():
    for i in range(10):
        shapes = ["원", "삼각형", "사각형", "오각형"]
        colors = ["red", "yellow", "blue", "green", "purple"]
        t.speed(0)

        shape = r.choice(shapes)
        color = r.choice(colors)
        size = r.randint(10, 100)
        x = r.randint(-200, 200)
        y = r.randint(-200, 200)

        t.penup()
        t.goto(x, y)
        t.pendown()

        t.color("black", color)
        t.begin_fill()

        if shape == "원":
            t.circle(size)
        elif shape == "삼각형":
            for _ in range(3):
                t.forward(size)
                t.left(120)
        elif shape == "사각형":
            for _ in range(4):
                t.forward(size)
                t.left(90)
        elif shape == "오각형":
            for _ in range(5):
                t.forward(size)
                t.left(72)
        t.end_fill()
    t.done()


# ------------------------------------------------------------
# [메인 실행부]
# ------------------------------------------------------------
if __name__ == "__main__":
    while True:
        print("\n==============================")
        print("  🐍 파이썬 실습 프로그램 모음")
        print("==============================")
        print("1. 산수 퀴즈")
        print("2. 아이디 확인")
        print("3. 학년별 수석 학생 구하기")
        print("4. 도형 그리기 (turtle)")
        print("5. 종료")
        choice = input("번호를 선택하세요 : ")

        if choice == '1':
            quiz_game()
        elif choice == '2':
            id_check()
        elif choice == '3':
            top_students()
        elif choice == '4':
            draw_shapes()
        elif choice == '5':
            print("프로그램을 종료합니다.")
            break
        else:
            print("잘못된 입력입니다. 다시 선택하세요.")
